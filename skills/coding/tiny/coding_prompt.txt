document.querySelectorAll('[data-filter]').forEach(btn => {
  btn.addEventListener('click', () => {
    const f = btn.dataset.filter;
    document.querySelectorAll('[data-category]').forEach(card => {
      card.style.display = (f === 'all' || card.dataset.category === f) ? '' : 'none';
    });
  });
});

// Live search
document.getElementById('search-input').addEventListener('input', function() {
  const q = this.value.toLowerCase();
  document.querySelectorAll('[data-category]').forEach(card => {
    card.style.display = card.innerText.toLowerCase().includes(q) ? '' : 'none';
  });
});

// Spec modal
function openModal(btn) {
  const card = btn.closest('[data-category]');
  document.getElementById('modal-content').innerHTML = card.querySelector('.specs-data').innerHTML;
  document.getElementById('spec-modal').classList.remove('hidden');
}
function closeModal() { document.getElementById('spec-modal').classList.add('hidden'); }
